75 enum class LogType
76 {
77 Message,
78 Warning,
79 Error,
80 Critical
81 };
82
83 static bool Init();
84
85 template <typename... Args>
86 static inline void LogMessage(LogType type, std::string_view message, Args&&... args)
87 {
88 std::string formattedMsg = std::vformat(message, std::make_format_args(args...));
89
90 switch (type)
91 {
92 case LogType::Message: LOG_MESSAGE(formattedMsg); break;
93 case LogType::Warning: LOG_WARNING(formattedMsg); break;
94 case LogType::Error: LOG_ERROR(formattedMsg); break;
95 case LogType::Critical: LOG_FATAL(formattedMsg); break;
96 };
97 };
98
99 static std::shared_ptr<spdlog::logger> GetEngineLogger() { return s_EngineLogger; }
100 static std::shared_ptr<spdlog::logger> GetGameLogger() { return s_GameLogger; }
101 static std::shared_ptr<spdlog::logger> GetEditorLogger() { return s_EditorLogger; }
102
103 private:
104 static std::shared_ptr<spdlog::logger> s_EngineLogger;
105 static std::shared_ptr<spdlog::logger> s_GameLogger;
106 static std::shared_ptr<spdlog::logger> s_EditorLogger;
107 };
