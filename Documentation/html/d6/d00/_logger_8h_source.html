113 template <typename... Args>
114 static inline void LogMessage(LogType type, std::string_view message, Args&&... args)
115 {
116 std::string formattedMsg = std::vformat(message, std::make_format_args(args...));
117
118 switch (type)
119 {
120 case LogType::Message: LOG_MESSAGE(formattedMsg); break;
121 case LogType::Warning: LOG_WARNING(formattedMsg); break;
122 case LogType::Error: LOG_ERROR(formattedMsg); break;
123 case LogType::Critical: LOG_FATAL(formattedMsg); break;
124 };
